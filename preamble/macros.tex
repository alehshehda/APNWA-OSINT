% Makra inline: \code (czysty monospace), \hash (łamany po każdym znaku)
% Obsługują polskie znaki UTF-8, ścieżki Windows (\) oraz _, %, #, &, $, ^, ~ bez escapowania.

\tl_new:N \l__report_seg_tl
\quark_new:N \q__report_stop

\cs_new_protected:Npn \__report_set_inline_catcodes:
  {
    \catcode 92  = 12\relax % \
    \catcode 95  = 12\relax % _
    \catcode 37  = 12\relax % %
    \catcode 35  = 12\relax % #
    \catcode 38  = 12\relax % &
    \catcode 36  = 12\relax % $
    \catcode 94  = 12\relax % ^
    \catcode 126 = 12\relax % ~
  }

\cs_new_protected:Npn \__report_flush_seg:
  {
    \tl_if_empty:NF \l__report_seg_tl
      {
        \l__report_seg_tl
        \tl_clear:N \l__report_seg_tl
      }
  }

\cs_new_protected:Npn \__report_emit_break:
  {
    \__report_flush_seg:
    \allowbreak
  }

\cs_new_protected:Npn \__report_walk_loop:n #1
  {
    \tl_if_head_is_space:nTF {#1}
      { \__report_walk_space:w . #1 \q_stop }
      {
        \tl_if_head_is_group:nTF {#1}
          { \__report_walk_group:nw #1 \q_stop }
          { \__report_walk_token:Nw #1 \q_stop }
      }
  }

\cs_new_protected:Npn \__report_walk_space:w . ~ #1 \q_stop
  {
    \__report_flush_seg:
    \space
    \allowbreak
    \__report_walk_loop:n {#1}
  }

\cs_new_protected:Npn \__report_walk_group:nw #1 #2 \q_stop
  {
    \tl_put_right:Nn \l__report_seg_tl { \textbraceleft }
    \__report_walk_loop:n { #1 \q__report_stop }
    \tl_put_right:Nn \l__report_seg_tl { \textbraceright }
    \__report_walk_loop:n {#2}
  }

\cs_new_protected:Npn \__report_walk_token:Nw #1 #2 \q_stop
  {
    \token_if_eq_meaning:NNTF #1 \q__report_stop
      { }
      {
        \__report_handle_token:N #1
        \__report_walk_loop:n {#2}
      }
  }

\cs_new_protected:Npn \__report_handle_token:N #1
  {
    \token_if_cs:NTF #1
      { \__report_handle_cs:N #1 }
      {
        \token_if_active:NTF #1
          {
            \int_compare:nNnTF { `#1 } = { 126 }
              { \tl_put_right:Nn \l__report_seg_tl { \textasciitilde } }
              { \tl_put_right:Nn \l__report_seg_tl {#1} }
          }
          { \__report_handle_char:N #1 }
      }
  }

\cs_new_protected:Npn \__report_handle_cs:N #1
  {
    \token_if_eq_meaning:NNTF #1 \{
      { \tl_put_right:Nn \l__report_seg_tl { \textbraceleft } }
      {
        \token_if_eq_meaning:NNTF #1 \}
          { \tl_put_right:Nn \l__report_seg_tl { \textbraceright } }
          {
            \token_if_eq_meaning:NNTF #1 \_
              {
                \tl_put_right:Nn \l__report_seg_tl { \_ }
                \__report_emit_break:
              }
              {
                \token_if_eq_meaning:NNTF #1 \%
                  { \tl_put_right:Nn \l__report_seg_tl { \% } }
                  {
                    \token_if_eq_meaning:NNTF #1 \#
                      { \tl_put_right:Nn \l__report_seg_tl { \# } }
                      {
                        \token_if_eq_meaning:NNTF #1 \&
                          { \tl_put_right:Nn \l__report_seg_tl { \& } }
                          {
                            \token_if_eq_meaning:NNTF #1 \$
                              { \tl_put_right:Nn \l__report_seg_tl { \$ } }
                              {
                                \token_if_eq_meaning:NNTF #1 \textbackslash
                                  {
                                    \tl_put_right:Nn \l__report_seg_tl { \textbackslash }
                                    \__report_emit_break:
                                  }
                                  {
                                    \token_if_eq_meaning:NNTF #1 \ %
                                      {
                                        \__report_flush_seg:
                                        \bool_if:NTF \l__report_inline_bg_bool
                                          { \__report_codebox:n { \space } }
                                          { \space }
                                        \allowbreak
                                      }
                                      {
                                        \tl_put_right:Nn \l__report_seg_tl { \textbackslash }
                                        \__report_emit_break:
                                        \tl_put_right:Nx \l__report_seg_tl { \cs_to_str:N #1 }
                                      }
                                  }
                              }
                          }
                      }
                  }
              }
          }
      }
  }

\cs_new_protected:Npn \__report_handle_char:N #1
  {
    \int_case:nnF { `#1 }
      {
        { 92 }  { \tl_put_right:Nn \l__report_seg_tl { \textbackslash } \__report_emit_break: }
        { 47 }  { \tl_put_right:Nn \l__report_seg_tl { / } \__report_emit_break: }
        { 46 }  { \tl_put_right:Nn \l__report_seg_tl { . } \__report_emit_break: }
        { 95 }  { \tl_put_right:Nn \l__report_seg_tl { \_ } \__report_emit_break: }
        { 45 }  { \tl_put_right:Nn \l__report_seg_tl { { - \kern 0pt } } \__report_emit_break: }
        { 37 }  { \tl_put_right:Nn \l__report_seg_tl { \% } }
        { 35 }  { \tl_put_right:Nn \l__report_seg_tl { \# } }
        { 38 }  { \tl_put_right:Nn \l__report_seg_tl { \& } }
        { 36 }  { \tl_put_right:Nn \l__report_seg_tl { \$ } }
        { 94 }  { \tl_put_right:Nn \l__report_seg_tl { \textasciicircum } }
        { 126 } { \tl_put_right:Nn \l__report_seg_tl { \textasciitilde } }
        { 60 }  { \tl_put_right:Nn \l__report_seg_tl { { < \kern 0pt } } }
        { 62 }  { \tl_put_right:Nn \l__report_seg_tl { { > \kern 0pt } } }
        { 96 }  { \tl_put_right:Nn \l__report_seg_tl { { ` \kern 0pt } } }
        { 39 }  { \tl_put_right:Nn \l__report_seg_tl { { ' \kern 0pt } } }
      }
      { \tl_put_right:Nn \l__report_seg_tl {#1} }
  }

\cs_new_protected:Npn \__report_inline_exec:n #1
  {
    \upshape\ttfamily
    \tl_clear:N \l__report_seg_tl
    \tl_if_empty:nF {#1}
      {
        \__report_walk_loop:n { #1 \q__report_stop }
        \__report_flush_seg:
      }
    \group_end:
  }

  }

  {

    \__report_hash_walk:n {#2}
  }

\cs_new_protected:Npn \__report_hash_token:Nw #1 #2 \q_stop
  {
    \token_if_eq_meaning:NNTF #1 \q__report_stop
      { }
      {
        \token_if_cs:NTF #1
          {
            \textbackslash\allowbreak
            \cs_to_str:N #1\allowbreak
          }
          {
            \token_if_active:NTF #1
              {
                \int_compare:nNnTF { `#1 } = { 126 }
                  { \textasciitilde\allowbreak }
                  { #1 }
              }
              {
                \int_case:nnF { `#1 }
                  {
                    { 92 }  { \textbackslash\allowbreak }
                    { 95 }  { \_\allowbreak }
                    { 37 }  { \%\allowbreak }
                    { 35 }  { \#\allowbreak }
                    { 38 }  { \&\allowbreak }
                    { 36 }  { \$\allowbreak }
                    { 94 }  { \textasciicircum\allowbreak }
                    { 126 } { \textasciitilde\allowbreak }
                    { 45 }  { { - \kern 0pt }\allowbreak }
                  }
                  { #1\allowbreak }
              }
          }
        \__report_hash_walk:n {#2}
      }
  }

\makeatletter
\newcommand{\UnnumberedChapter}[1]{%
  \clearpage\phantomsection
  \if@mainmatter
    \addcontentsline{toc}{chapter}{#1}%
  \else
    \protected@write\@auxout
      {\let\label\@gobble \let\index\@gobble \let\glossary\@gobble}%
      {\string\@writefile{toc}{\protect\contentsline{chapter}{#1}{}{\@currentHref}}}%
  \fi
  \chapter*{#1}%
}
