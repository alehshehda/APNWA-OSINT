% Nagłówki i stopki (fancyhdr) oraz helper \BackList dla spisów końcowych.

\makeatletter
\newcommand{\HeaderChapterLabel}{\@chapapp\ \thechapter}
\makeatother

\newcommand{\HeaderInstitution}{%
  \footnotesize\color{dgray}%
  \IfNotEmpty{\University}{\University}%
  \IfNotEmpty{\University}{\IfNotEmpty{\Faculty}{\ \textendash\ }}%
  \IfNotEmpty{\Faculty}{\Faculty}%
}

\newcommand{\HeaderInstitutionLine}{%
  \IfEmpty{\University}{%
    \IfNotEmpty{\Faculty}{\HeaderInstitution\par\nointerlineskip}%
  }{%
    \HeaderInstitution\par\nointerlineskip
  }%
}

\newcommand{\HeaderOddBox}[1]{%
  \parbox[b]{\headwidth}{%
    \raggedright\noindent\HeaderInstitutionLine
    \noindent\makebox[\headwidth]{{\color{Accent}\hrulefill}\hspace{1em}\large #1}}}

\newcommand{\HeaderEvenBox}[1]{%
  \parbox[b]{\headwidth}{%
    \raggedleft\noindent\HeaderInstitutionLine
    \noindent\makebox[\headwidth]{\large #1\hspace{1em}{\color{Accent}\hrulefill}}}}

\newcommand{\FooterSubject}{\IfEmpty{\ShortSubject}{\Subject}{\ShortSubject}}

\makeatletter
\newcommand{\FooterSetup}{%
  \fancyhf{}%
  \if@twoside
    \fancyfoot[LE,RO]{\thepage}%
    \fancyfoot[LO,RE]{\small\FooterSubject}%
  \else
    \fancyfoot[R]{\thepage}%
    \fancyfoot[L]{\small\FooterSubject}%
  \fi
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0.2pt}%
}

\fancypagestyle{frontstyle}{%
  \fancyhf{}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}%
}

\fancypagestyle{chapterfirst}{\FooterSetup}
\fancypagestyle{plain}{\FooterSetup}

\newcommand{\DefineHeaderStyle}[2]{%
  \fancypagestyle{#1}{%
    \FooterSetup
    \if@twoside
      \fancyhead[RO]{\HeaderOddBox{#2}}%
      \fancyhead[LE]{\HeaderEvenBox{#2}}%
    \else
      \fancyhead[R]{\HeaderOddBox{#2}}%
    \fi
  }%
}
\makeatother

\DefineHeaderStyle{mainstyle}{\HeaderChapterLabel}
\DefineHeaderStyle{bibstyle}{\bibname}
\DefineHeaderStyle{lofstyle}{\listfigurename}
\DefineHeaderStyle{lotstyle}{\listtablename}
\DefineHeaderStyle{lolstyle}{\lstlistlistingname}
\DefineHeaderStyle{loastyle}{\listalgorithmcfname}

% Spis końcowy z wpisem do spisu treści i dedykowanym stylem stron
\newcommand{\BackList}[3]{%
  \clearpage\phantomsection
  \pagestyle{#1}%
  \addcontentsline{toc}{chapter}{#2}%
  #3%
}
